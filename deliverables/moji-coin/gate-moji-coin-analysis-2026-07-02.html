def moji_coin_detector(kline_5m, lookback_hours=6): bars = lookback_hours * 12 # 72 bars pre_df = kline_5m.tail(bars) score = 0; triggered = [] # R1: Volume Spike vol_ratio = last_1h_volume / earlier_5h_volume if vol_ratio >= 1.5: score += 25; triggered.append('R1') # R2: Price Staircase stair_pct = count(close[i] >= close[i-1]) / 5 if stair_pct >= 0.80: score += 20; triggered.append('R2') # R3: Volume Staircase vol_stair_pct = count(vol[i] >= vol[i-1]) / 5 if vol_stair_pct >= 0.60: score += 20; triggered.append('R3') # R4: Range Expansion range_ratio = recent_range_avg / earlier_range_avg if range_ratio >= 1.20: score += 15; triggered.append('R4') # R5: Mild Accumulation cum_change = (latest_close - first_open) / first_open * 100 if cum_change >= 4.31: score += 10; triggered.append('R5') # R6: Volatility Compression volatility = log_returns.std() * sqrt(288) if volatility <= 0.1009: score += 10; triggered.append('R6') if score >= 80 and len(triggered) >= 3: level = 'STRONG' elif score >= 60 and len(triggered) >= 2: level = 'MEDIUM' elif score >= 40: level = 'WEAK' else: level = 'NONE' return {'score': score, 'level': level, 'triggered_rules': triggered}
